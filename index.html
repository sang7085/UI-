JS (jQuery) — 열기 / 닫기 / 딤 클릭 / ESC

                $(document).on("click", "[data-popup_open]", function () {
                  $($(this).data("popup_open")).addClass("is_open");
                });
                $(document).on("click", "[data-popup_close]", function () {
                  $(this).closest(".popup_layer").removeClass("is_open");
                });
                $(document).on("click", ".popup_layer", function (e) {
                  if (e.target === this) $(this).removeClass("is_open");
                });
                $(document).on("keydown", function (e) {
                  if (e.key === "Escape") $(".popup_layer.is_open").removeClass("is_open");
                });

CSS

                  position: fixed;
                  left: 50%;
                  bottom: 32px;
                  z-index: 1100;
                  padding: 12px 20px;
                  border-radius: 999px;
                  background: rgba(34, 34, 34, 0.92);
                  color: #fff;
                  font-size: 14px;
                  white-space: nowrap;
                  opacity: 0;
                  visibility: hidden;
                  transform: translate(-50%, 16px);
                  transition: opacity 0.25s ease, transform 0.25s ease, visibility 0.25s ease;
                }
                .popup_toast.is_show {
                  opacity: 1;
                  visibility: visible;
                  transform: translate(-50%, 0);
                }

JS (jQuery) — 2초 뒤 자동으로 사라짐

                $(document).on("click", "[data-toast_open]", function () {
                  var $toast = $($(this).data("toast_open"));
                  clearTimeout($toast.data("timer"));
                  $toast.addClass("is_show");
                  $toast.data("timer", setTimeout(function () {
                    $toast.removeClass("is_show");
                  }, 2000));
                });
